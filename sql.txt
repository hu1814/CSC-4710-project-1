-- 1. Create the Users table (sql/init.sql)
CREATE TABLE Users (
    username VARCHAR(50) PRIMARY KEY,
    password VARCHAR(255) NOT NULL,
    firstname VARCHAR(50),
    lastname VARCHAR(50),
    salary DECIMAL(10,2),
    age INTEGER,
    registerday DATETIME DEFAULT CURRENT_TIMESTAMP,
    signintime DATETIME NULL
);

-- 2. Backend startup database connectivity check
SELECT username FROM Users LIMIT 0;

-- 3. User registration (password placeholder receives an Argon2id hash)
INSERT INTO Users (username, password, firstname, lastname, salary, age)
VALUES (?, ?, ?, ?, ?, ?);

-- 4. Sign-in: look up stored password hash by username
SELECT password FROM Users WHERE username = ?;

-- 5. Sign-in: record latest successful sign-in
UPDATE Users SET signintime = CURRENT_TIMESTAMP WHERE username = ?;

-- 6. Fetch user by username (also used after registration/sign-in)
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE username = ?;

-- 7. Search users by first name
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE firstname = ? ORDER BY registerday, username;

-- 8. Search users by last name
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE lastname = ? ORDER BY registerday, username;

-- 9. Search users by both first and last name
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE firstname = ? AND lastname = ? ORDER BY registerday, username;

-- 10. Search users in a salary range (inclusive)
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE salary BETWEEN ? AND ? ORDER BY salary, username;

-- 11. Search users in an age range (inclusive)
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE age BETWEEN ? AND ? ORDER BY age, username;

-- 12. Search users who registered after a specified user.
-- Backend first fetches the reference user's registerday with query 6.
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE registerday > ? ORDER BY registerday, username;

-- 13. Search users who have never signed in
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE signintime IS NULL ORDER BY registerday, username;

-- 14. Search users who registered on the same day as a specified user.
-- Backend first fetches the reference user's registerday with query 6.
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday >= DATE(?) AND registerday < DATE(?) + INTERVAL 1 DAY
ORDER BY registerday, username;

-- 15. Search users registered today
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday >= CURRENT_DATE() AND registerday < CURRENT_DATE() + INTERVAL 1 DAY
ORDER BY registerday, username;
